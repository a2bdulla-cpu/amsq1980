def is_market_open() -> bool:
    tz = pytz.timezone('America/New_York')
    ny_time = datetime.now(tz)
    if ny_time.weekday() >= 5:
        return False
    market_start = 9 * 60 + 30
    market_end = 16 * 60
    current_minutes = ny_time.hour * 60 + ny_time.minute
    return market_start <= current_minutes <= market_end

def fetch_live_opportunities() -> list:
    cfg = load_scanner_settings()
    symbols_to_scan = get_watchlist()
    found_opportunities = []

    for symbol in symbols_to_scan:
        try:
            stock = yf.Ticker(symbol)
            info = stock.info
            price = info.get('regularMarketPrice') or info.get('previousClose') or 0
            float_shares = info.get('floatShares', 0) or 0

            if not (cfg["min_price"] <= price <= cfg["max_price"]):
                continue
            if float_shares > 0 and not (cfg["min_float"] <= float_shares <= cfg["max_float"]):
                continue

            df = stock.history(period="5d", interval="15m")
            if df.empty or len(df) < 20:
                continue

            avg_vol = df['Volume'].rolling(20).mean().iloc[-1]
            current_vol = df['Volume'].iloc[-1]
            rvol = round(current_vol / (avg_vol if avg_vol > 0 else 1), 2)

            if rvol < cfg["min_rvol"]:
                continue

            portfolio_balance = 400.0
            allocated_funds = portfolio_balance * cfg["allocation_pct"]
            shares = int(allocated_funds / price) if price > 0 else 0
            if shares == 0:
                continue

            buy_comm = max(shares * 0.0049, 0.99)
            cost_with_comm = round((shares * price) + buy_comm, 2)

            found_opportunities.append({
                "symbol": symbol,
                "price": price,
                "session_type": "Live Market ??",
                "float_m": round(float_shares / 1_000_000, 2) if float_shares else 0.0,
                "rvol": rvol,
                "shares": shares,
                "cost_with_comm": cost_with_comm,
                "sl": round(price * (1 - cfg["risk_pct"]), 2),
                "tp1": round(price * 1.10, 2)
            })
        except Exception:
            continue
    return found_opportunities
